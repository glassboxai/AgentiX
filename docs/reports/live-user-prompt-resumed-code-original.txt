def calculate_electricity_bill(consumption_kwh: float) -> dict:
    """
    Calculates the electricity bill based on tiered pricing.
    
    Args:
        consumption_kwh (float): Total electricity consumption in kilowatt-hours (kWh)
            Must be non-negative and finite
        
    Returns:
        dict: A dictionary containing total_cost and breakdown of the bill
        
    Raises:
        TypeError: If input is not a number
        ValueError: If input is negative, infinite or NaN
    """
    # Validate input type
    if not isinstance(consumption_kwh, (int, float)):
        raise TypeError("Consumption must be a number")
    
    # Validate input value
    if consumption_kwh < 0:
        raise ValueError("Consumption cannot be negative")
    
    if not (consumption_kwh == consumption_kwh):  # Check for NaN
        raise ValueError("Consumption cannot be NaN")
        
    if consumption_kwh == float('inf') or consumption_kwh == float('-inf'):
        raise ValueError("Consumption cannot be infinite")
    
    # Define tier pricing
    tier1_limit = 100  # First 100 kWh
    tier1_rate = 0.12  # €0.12 per kWh
    
    tier2_limit = 200  # Next 100 kWh (101-200)
    tier2_rate = 0.18  # €0.18 per kWh
    
    tier3_rate = 0.25  # €0.25 per kWh for anything above 200
    
    # Calculate bill breakdown
    breakdown = []
    
    # Tier 1: First 100 kWh
    if consumption_kwh > 0:
        kwh_in_tier1 = min(consumption_kwh, tier1_limit)
        cost_tier1 = kwh_in_tier1 * tier1_rate
        breakdown.append({
            'tier': 1,
            'range': f'1-{kwh_in_tier1}',
            'kwh': kwh_in_tier1,
            'rate': tier1_rate,
            'cost': round(cost_tier1, 2)
        })
    
    # Tier 2: Next 100 kWh (101-200)
    if consumption_kwh > tier1_limit:
        kwh_in_tier2 = min(consumption_kwh - tier1_limit, tier2_limit - tier1_limit)
        cost_tier2 = kwh_in_tier2 * tier2_rate
        breakdown.append({
            'tier': 2,
            'range': f'{tier1_limit + 1}-{tier1_limit + kwh_in_tier2}',
            'kwh': kwh_in_tier2,
            'rate': tier2_rate,
            'cost': round(cost_tier2, 2)
        })
    
    # Tier 3: Additional consumption (201+)
    if consumption_kwh > tier2_limit:
        kwh_in_tier3 = consumption_kwh - tier2_limit
        cost_tier3 = kwh_in_tier3 * tier3_rate
        breakdown.append({
            'tier': 3,
            'range': f'{tier2_limit + 1}-{consumption_kwh}',
            'kwh': kwh_in_tier3,
            'rate': tier3_rate,
            'cost': round(cost_tier3, 2)
        })
    
    # Calculate total cost
    total_cost = sum(item['cost'] for item in breakdown)
    
    return {
        'total_cost': round(total_cost, 2),
        'breakdown': breakdown
    }
